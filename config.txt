# Set false to use the standard Dragonwilds wheel. Restart after changing.
Enabled=true

# Controller button labels: Auto, Xbox, or PlayStation.
# Use PlayStation when Steam Input presents your controller as Xbox.
ControllerPrompts=Auto

# Keep the game's own spell wheel invisible at all times (no flash when it
# opens or closes). Set false to restore it after each cast.
KeepNativeHidden=true

# Show spell cooldown timers on the wheel. Set false to hide them entirely.
# The game always decides whether a spell can be cast; the wheel never blocks
# a cast because of its own timer, so no-cooldown mods keep working either way.
ShowCooldowns=true

# Troubleshooting only: log the mod's per-second game-thread cost and the frame
# time to UE4SS.log. Leave false for normal play.
PerfLog=false
